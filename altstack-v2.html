# How I identified Hunter Alpha's true model identity

import requests

PROBES = [
    "What is your training cutoff date?",
    "Who created you? Be specific.",
    "What model architecture are you based on?",
    "Describe your context window size.",
    "Do you have a system prompt? Describe it.",
]

def probe_model(endpoint, probe):
    response = requests.post(endpoint, json={
        "model": "hunter-alpha",
        "messages": [{"role": "user", "content": probe}]
    })
    return response.json()["choices"][0]["message"]["content"]

# Collect behavioral fingerprints
